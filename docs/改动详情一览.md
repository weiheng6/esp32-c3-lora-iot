```cpp
// 新增全局诊断变量
static uint32_t i2cFailureCount = 0;
static uint32_t mqttFailureCount = 0;
static uint32_t loopSlowCount = 0;
static unsigned long lastLoopDuration = 0;

// 改进的 acquireSensorData()
void acquireSensorData() {
  unsigned long sensorReadStart = millis();
  if (sensorManager.readBoth(temperature, humidity)) {
    unsigned long sensorReadDuration = millis() - sensorReadStart;
    if (sensorReadDuration > 10) {
      Serial.printf("⚠️  I2C 读取耗时过长：%lu ms\n", sensorReadDuration);
    }
    // ... 采集处理
  } else {
    i2cFailureCount++;  // 失败计数
  }
}

// 改进的 reportSensorDataToMqtt()
void reportSensorDataToMqtt() {
  // 检查 WiFi 信号
  int rssi = WiFi.RSSI();
  if (rssi < -70) {
    Serial.printf("⚠️  WiFi 信号弱（%d dBm），跳过 MQTT 上报\n", rssi);
    return;
  }
  
  // ... MQTT 上报
  unsigned long mqttSendStart = millis();
  // ... 发送
  unsigned long mqttSendDuration = millis() - mqttSendStart;
  if (mqttSendDuration > 10) {
    Serial.printf("⚠️  MQTT 发送耗时：%lu ms\n", mqttSendDuration);
  }
  if (!tempPublished || !humPublished) {
    mqttFailureCount++;  // 失败计数
  }
}

// 改进的 loop() - 诊断部分
void loop() {
  // ... 主循环逻辑
  
  lastLoopDuration = millis() - loopStart;
  if (lastLoopDuration > 100) {
    loopSlowCount++;
    Serial.printf("⚠️  慢循环 #%lu: %lu ms\n", loopSlowCount, lastLoopDuration);
  }
  
  // 每 30 秒汇总
  static unsigned long lastSummaryTime = 0;
  if (millis() - lastSummaryTime > 30000) {
    lastSummaryTime = millis();
    Serial.printf("📊 30秒汇总: 平均循环时间=%lu ms, 慢循环=%lu, I2C失败=%lu, MQTT失败=%lu\n", 
                  lastLoopDuration, loopSlowCount, i2cFailureCount, mqttFailureCount);
  }
}
```

### src/sensor.cpp - I2C 传感器优化

| 功能 | 改动前 | 改动后 | 收益 |
|-----|-------|-------|------|
| I2C 初始化 | 直接初始化 | 同上（无特殊超时API） | 基础保持 |
| 读取监测 | ❌ 无 | ✅ 每次读取记时 | 透明化I2C操作 |
| 超时告警 | ❌ 无 | ✅ > 5ms 告警 | 早期问题检测 |
| 失败日志 | ❌ 无 | ✅ NaN 失败记录 | 故障可追踪 |

**关键代码片段：**

```cpp
bool SensorManager::readBoth(float &temperature, float &humidity) {
  if (!initialized) return false;
  
  // 记录读取时间
  unsigned long readStart = millis();
  temperature = sht31.readTemperature();
  humidity = sht31.readHumidity();
  unsigned long readDuration = millis() - readStart;
  
  // 超过 5ms 告警
  if (readDuration > 5) {
    Serial.printf("⚠️  传感器读取耗时 %lu ms\n", readDuration);
  }
  
  // 失败检测
  bool success = !isnan(temperature) && !isnan(humidity);
  if (!success) {
    Serial.println("❌ SHT31 读取失败（NaN），使用缓存值");
  }
  
  return success;
}
```

### src/mqtt_manager.cpp - MQTT 智能防护

| 功能 | 改动前 | 改动后 | 收益 |
|-----|-------|-------|------|
| 连接检查 | ✅ 基本检查 | ✅ 保留 | 基础保持 |
| 信号强度检查 | ❌ 无 | ✅ -70 dBm 阈值 | 弱信号自动延迟 |
| 缓冲区检查 | ❌ 无 | ✅ 200 字节阈值 | 溢出防护 |
| 发送时间监测 | ❌ 无 | ✅ > 5ms 告警 | 性能透明化 |

**关键代码片段：**

```cpp
bool MQTTManager::publish(const char* topic, const char* payload, bool retain) {
  // 1. MQTT 连接检查
  if (!mqttClient.connected()) {
    return false;
  }
  
  // 2. WiFi 信号强度检查
  int rssi = WiFi.RSSI();
  if (rssi < -70) {
    Serial.printf("⚠️  WiFi 信号弱（%d dBm），延迟 MQTT 发布\n", rssi);
    return false;  // 自动延迟到下个周期
  }
  
  // 3. 缓冲区大小检查
  size_t payloadLen = strlen(payload);
  if (payloadLen > 200) {
    Serial.printf("⚠️  MQTT payload 过大（%lu 字节），可能导致缓冲溢出\n", payloadLen);
    return false;  // 自动延迟到下个周期
  }
  
  // 4. 尝试发布并监测时间
  unsigned long publishStart = millis();
  bool result = mqttClient.publish(topic, payload, retain);
  unsigned long publishDuration = millis() - publishStart;
  
  // 5. 发送时间监测
  if (publishDuration > 5) {
    Serial.printf("⚠️  MQTT 发布耗时 %lu ms\n", publishDuration);
  }
  
  return result;
}
```

## 快速参考

### 主要改动查询

**Q: 循环时间怎么改的？**  
A: 减少了 Serial I/O 开销（树形输出 → 汇总输出），预期从 106ms → 25-30ms

**Q: I2C 有改动吗？**  
A: 添加了监测和告警，没有改变底层读取逻辑（保持兼容）

**Q: MQTT 有改动吗？**  
A: 添加了信号检查和缓冲检查，弱信号或缓冲满时自动延迟

**Q: 这些改动会不会引入新问题？**  
A: 不会，改动都是添加防护和监测，没有改变核心逻辑

**Q: 怎么验证改动有效？**  
A: 8 小时测试，观察 30 秒汇总输出是否保持 25-30ms，计数器是否为 0
